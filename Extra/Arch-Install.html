Arch Installation Steps (Advanced)

Download Arch linux iso file from https://archlinux.org
Burn the downloaded iso into a pendrive or SD card using Rufus
Go into BIOS and disable Secure Boot

Boot Into the Pendrive

Type the following commands

# setfont ter-132n # iwctl # device list # station wlan0 get-networks # station wlan0 connect WifiName Passphrase: type the password # exit # clear # pacman -Sy # pacman -Sy archlinux-keyring # lsblk # cfdisk /dev/nvme0n1

create 800M type EFI system partition
create 8G type Linux swap partition
create remaining type Linux filesystem partition
write the changes to the disk and quit

# mkfs.fat -F32 /dev/efi_partition_name # mkfs.ext4 /dev/root_partition_name # mkswap /dev/swap_partition_name # mount /dev/root_partition_name /mnt # mkdir /mnt/boot # mount /dev/efi_partition_name /mnt/boot # swapon /dev/swap_partition_name

Check if they are mounted properly using lsblk command

# pacstrap -i /mnt base base-devel linux linux-firmware git sudo fastfetch htop intel-ucode nano bluez bluez-utils networkmanager # genfstab -U /mnt >> /mnt /mnt/etc/fstab # cat /mnt/etc/fstab # arch-chroot /mnt # fastfetch # passwd

type the password

# useradd -m -g users -G wheel,storage,power,video,audio -s /bin/bash username # passwd username

type the password

# EDITOR=nano visudo

scroll and uncomment %wheel ALL=(ALL:ALL) ALL

Ctrl+O to save changes, Ctrl+X to exit

# su - username # sudo pacman -Syu # exit # ln -sf /usr/share/zoneinfo/ # ln -sf /usr/share/zoneinfo/Asia/Kolkata /etc/localtime # hwclock --systohc # nano /etc/locale.gen

scroll and uncomment en_US.UTF-8 UTF-8

# locale-gen # nano /etc/locale.conf

write LANG=en_US.UTF-8

# nano /etc/hostname

write archlinux

# nano /etc/hosts

type the following after 2 line gap

# pacman -S grub efibootmgr dosfstools mtools # grub-install --target=x86_64-efi --efi-directory=/boot --bootloader-id=GRUB # grub-mkconfig -o /boot/grub/grub.cfg # systemctl enable bluetooth # systemctl enable NetworkManager # exit # umount -lR /mnt # shutdown now

Now Remove the USB drive and start the PC
Boot into Arch Linux and then login with your username

# setfont -d # nmcli dev status # nmcli radio wifi on # nmcli dev wifi list # sudo nmcli dev wifi connect WifiName password "WifiPass" # sudo pacman -Sy # sudo pacman -Syu # sudo pacman -Sy flatpak # sudo pacman -S hyprland dunst kitty xdg-desktop-portal-hyprland qt5-wayland qt6-wayland pipewire thunar vlc clang cargo firefox noto-fonts noto-fonts-emoji ttf-dejavu ttf-font-awesome hyprlang hyprcursor dunst polkit-kde-agent wireplumber libinput sddm # sudo nano /etc/default/grub

set grub timeout to 20 sec
uncomment grub disable os prober = false
write the changes and exit
